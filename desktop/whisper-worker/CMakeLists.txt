cmake_minimum_required(VERSION 3.16)
project(studio_whisper LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT WHISPER_SOURCE)
  get_filename_component(WHISPER_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../.build/desktop-whisper" ABSOLUTE)
endif()
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
add_subdirectory("${WHISPER_SOURCE}" whisper)
add_executable(studio-whisper main.cpp)
target_link_libraries(studio-whisper PRIVATE whisper)
